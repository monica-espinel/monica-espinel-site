{{- /* Responsive image for a site-relative upload path. Params: src, alt, sizes, class, eager (bool), maxWidth (default 1200). Uploads (mounted as assets) get WebP sizes in a srcset; other images are output as they are. Animated GIFs are left untouched. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $sizes := .sizes | default "(max-width: 1240px) 100vw, 1200px" -}} {{- $max := .maxWidth | default 1200 -}} {{- $loading := cond (.eager | default false) "eager" "lazy" -}} {{- $res := "" -}} {{- if and (hasPrefix $src "/wp-content/uploads/") (not (hasSuffix (lower $src) ".gif")) -}} {{- $res = resources.Get (strings.TrimPrefix "/" $src) -}} {{- end -}} {{- if $res -}} {{- $set := slice -}} {{- $fallback := $res -}} {{- range slice 400 800 1200 1600 2000 -}} {{- if and (le . $res.Width) (le . (mul $max 2)) -}} {{- $r := $res.Resize (printf "%dx webp" .) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink .) -}} {{- if le . $max }}{{ $fallback = $r }}{{ end -}} {{- end -}} {{- end -}} {{- if not $set }}{{ $fallback = $res.Resize (printf "%dx webp" $res.Width) }}{{ $set = slice (printf "%s %dw" $fallback.RelPermalink $res.Width) }}{{ end -}} {{- else -}} {{- end -}}

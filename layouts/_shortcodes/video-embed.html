{{/* WordPress embed block for YouTube or Vimeo, keeping the start time. */}} {{- $id := .Get "id" -}} {{- $start := .Get "start" -}} {{- $src := "" -}} {{- if eq (.Get "provider") "youtube" -}} {{- $src = printf "https://www.youtube-nocookie.com/embed/%s" $id -}} {{- with $start }}{{ $src = printf "%s?start=%s" $src . }}{{ end -}} {{- else -}} {{- $src = printf "https://player.vimeo.com/video/%s" $id -}} {{- with $start }}{{ $src = printf "%s#t=%s" $src . }}{{ end -}} {{- end -}}
